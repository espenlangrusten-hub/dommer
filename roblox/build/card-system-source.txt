Cards.ByRarity = {} :: { [string]: { Card } }
Cards.ByCategory = {} :: { [string]: { Card } }

for _, card in list do
	assert(Cards.ById[card.Id] == nil, `duplicate card id "{card.Id}"`)
	Rarities.get(card.Rarity)
	assert(Categories.exists(card.Category), `card "{card.Id}" has unknown category "{card.Category}"`)
	Cards.ById[card.Id] = card

	local byRarity = Cards.ByRarity[card.Rarity]
	if not byRarity then
		byRarity = {}
		Cards.ByRarity[card.Rarity] = byRarity
	end
	table.insert(byRarity, card)

	local byCategory = Cards.ByCategory[card.Category]
	if not byCategory then
		byCategory = {}
		Cards.ByCategory[card.Category] = byCategory
	end
	table.insert(byCategory, card)
end

function Cards.get(id: string): Card?
	return Cards.ById[id]
end

for _, rarity in Rarities.List do
	assert(Cards.ByRarity[rarity.Id], `rarity "{rarity.Id}" has no cards`)
end

for _, category in Categories.List do
	assert(Cards.ByCategory[category.Id], `category "{category.Id}" has no cards`)
end

return Cards

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")

local FOLDER_NAME = "CardRemotes"
local EVENT_NAMES = {
	"RequestRoll",
	"RollResult",
	"RollRejected",
	"SelectCard",
	"SelectionResult",
	"EquipCard",
	"ProfileSync",
	"Notice",
}

export type Remotes = {
	RequestRoll: RemoteEvent,
	RollResult: RemoteEvent,
	RollRejected: RemoteEvent,
	SelectCard: RemoteEvent,
	SelectionResult: RemoteEvent,
	EquipCard: RemoteEvent,
	ProfileSync: RemoteEvent,
	Notice: RemoteEvent,
}

local cached: Remotes? = nil

local function build(): Remotes
	local remotes = {}

	if RunService:IsServer() then
		local folder = ReplicatedStorage:FindFirstChild(FOLDER_NAME)
		if not folder then
			folder = Instance.new("Folder")
			folder.Name = FOLDER_NAME
			folder.Parent = ReplicatedStorage
		end

		for _, name in EVENT_NAMES do
			local event = folder:FindFirstChild(name)
			if not event then
				event = Instance.new("RemoteEvent")
				event.Name = name
				event.Parent = folder
			end
			remotes[name] = event
		end
	else
		local folder = ReplicatedStorage:WaitForChild(FOLDER_NAME)
		for _, name in EVENT_NAMES do
			remotes[name] = folder:WaitForChild(name)
		end
	end

	return (remotes :: any) :: Remotes
end

return {
	get = function(): Remotes
		if not cached then
			cached = build()
		end
		return (cached :: any) :: Remotes
	end,
}

local Cards = require(script.Parent.Cards)
local Config = require(script.Parent.Config)
local Rarities = require(script.Parent.Rarities)

type Card = Cards.Card
type Rarity = Rarities.Rarity

local RollLogic = {}

function RollLogic.pickRarity(rng: Random): Rarity
	local roll = rng:NextNumber(0, Rarities.TotalWeight)
	local running = 0
	for _, rarity in Rarities.List do
		running += rarity.Weight
		if roll <= running then
			return rarity
		end
	end

	return Rarities.List[#Rarities.List]
end

function RollLogic.pickCardOfRarity(rng: Random, rarityId: string): Card
	local pool = Cards.ByRarity[rarityId]
	assert(pool and #pool > 0, `no cards for rarity "{rarityId}"`)
	return pool[rng:NextInteger(1, #pool)]
end

function RollLogic.pickCard(rng: Random): Card
	return RollLogic.pickCardOfRarity(rng, RollLogic.pickRarity(rng).Id)
end

function RollLogic.rollHand(rng: Random, pityCounter: number): ({ Card }, number)
	local hand: { Card } = {}
	for _ = 1, Config.CardsPerRoll do
		table.insert(hand, RollLogic.pickCard(rng))
	end

	local minRank = Rarities.get(Config.PityMinimumRarity).Rank
	local best = 0
	for _, card in hand do
		best = math.max(best, Rarities.get(card.Rarity).Rank)
	end

	local pityActive = Config.PityRolls > 0
	if pityActive and best < minRank and pityCounter + 1 >= Config.PityRolls then

		local slot = rng:NextInteger(1, #hand)
		hand[slot] = RollLogic.pickCardOfRarity(rng, Config.PityMinimumRarity)
		best = minRank
	end

	local nextPity = if best >= minRank then 0 else pityCounter + 1
	return hand, nextPity
end

return RollLogic

function Theme.make(className: string, props: { [string]: any }, children: { any }?): any
	local instance = Instance.new(className)
	local parent = props.Parent
	for key, value in props do
		if key ~= "Parent" then
			(instance :: any)[key] = value
		end
	end
	if children then
		for _, child in children do
			child.Parent = instance
		end
	end

	if parent then
		instance.Parent = parent
	end
	return instance
end

local make = Theme.make

function Theme.corner(radius: number): any
	return make("UICorner", { CornerRadius = UDim.new(0, radius) })
end

function Theme.stroke(color: Color3, thickness: number, transparency: number?): UIStroke
	return make("UIStroke", {
		Color = color,
		Thickness = thickness,
		Transparency = transparency or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
		LineJoinMode = Enum.LineJoinMode.Round,
	})
end

function Theme.gradient(top: Color3, bottom: Color3, rotation: number?): any

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local CardSystem = ReplicatedStorage:WaitForChild("CardSystem")
local Cards = require(CardSystem:WaitForChild("Cards"))
local Config = require(CardSystem:WaitForChild("Config"))
local Rarities = require(CardSystem:WaitForChild("Rarities"))

local SAVE_ENABLED = true
local STORE_NAME = "CardInventory_v2"
local LOAD_RETRIES = 3

export type Record = {
	Counts: { [string]: number },
	Equipped: { [string]: string },
	Pity: number,
	Rolls: number,
}

export type Snapshot = {
	Counts: { [string]: number },
	Equipped: { string | boolean },
	Rolls: number,
}

local Inventory = {}

local store: DataStore? = nil
local storeWarned = false
local records: { [Player]: Record } = {}

if SAVE_ENABLED then
	local ok, result = pcall(function()
		return DataStoreService:GetDataStore(STORE_NAME)
	end)
	if ok then
		store = result
	else
		warn(`[CardSystem] DataStore unavailable, profiles are session only: {result}`)
	end
end

local function emptyRecord(): Record
	return { Counts = {}, Equipped = {}, Pity = 0, Rolls = 0 }
end

local function sanitise(raw: any): Record
	local record = emptyRecord()
	if typeof(raw) ~= "table" then
		return record
	end

	if typeof(raw.Counts) == "table" then
		for cardId, count in raw.Counts do

			if typeof(cardId) == "string" and typeof(count) == "number" and Cards.get(cardId) then
				record.Counts[cardId] = math.max(0, math.floor(count))
			end
		end
	end

	if typeof(raw.Equipped) == "table" then
		for slotKey, cardId in raw.Equipped do
			local slot = tonumber(slotKey)
			if
				slot
				and slot == math.floor(slot)
				and slot >= 1
				and slot <= Config.Hotbar.Slots
				and typeof(cardId) == "string"
				and Cards.get(cardId)
				and (record.Counts[cardId] or 0) > 0
			then
				record.Equipped[tostring(slot)] = cardId
			end
		end
	end

	for _, key in { "Pity", "Rolls" } do
		if typeof(raw[key]) == "number" then
			(record :: any)[key] = math.max(0, math.floor(raw[key]))
		end
	end
	return record
end

local function keyFor(player: Player): string
	return `player_{player.UserId}`
end

local function totalCards(record: Record): number
	local total = 0
	for _, count in record.Counts do
		total += count
	end
	return total
end

local function bestRarityName(record: Record): string
	local bestRank, bestName = 0, "-"
	for cardId, count in record.Counts do
		local card = Cards.get(cardId)
		if card and count > 0 then
			local rarity = Rarities.get(card.Rarity)
			if rarity.Rank > bestRank then
				bestRank, bestName = rarity.Rank, rarity.Name
			end
		end
	end
	return bestName
end

local function refreshStats(player: Player)
	local record = records[player]
	local stats = player:FindFirstChild("leaderstats")
	if not record or not stats then
		return
	end

	local cardsStat = stats:FindFirstChild("Cards") :: IntValue?
	if cardsStat then
		cardsStat.Value = totalCards(record)
	end
	local bestStat = stats:FindFirstChild("Best") :: StringValue?
	if bestStat then
		bestStat.Value = bestRarityName(record)
	end
end

function Inventory.snapshot(player: Player): Snapshot?
	local record = records[player]
	if not record then
		return nil
	end

	local counts = {}
	for cardId, count in record.Counts do
		counts[cardId] = count
	end

	local equipped: { string | boolean } = {}
	for slot = 1, Config.Hotbar.Slots do
		equipped[slot] = record.Equipped[tostring(slot)] or false
	end

	return {
		Counts = counts,
		Equipped = equipped,
		Rolls = record.Rolls,
	}
end

function Inventory.load(player: Player)
	local record = emptyRecord()

	if store then
		for attempt = 1, LOAD_RETRIES do
			local ok, result = pcall(function()
				return (store :: DataStore):GetAsync(keyFor(player))
			end)
			if ok then
				record = sanitise(result)
				break
			end
			if attempt == LOAD_RETRIES then
				if not storeWarned then
					storeWarned = true
					warn(`[CardSystem] profile load failed for {player.Name}: {result}`)
				end
			else
				task.wait(attempt)
			end
		end
	end

	if not player:IsDescendantOf(Players) then
		return
	end

	records[player] = record

	local stats = Instance.new("Folder")
	stats.Name = "leaderstats"

	local cardsStat = Instance.new("IntValue")
	cardsStat.Name = "Cards"
	cardsStat.Parent = stats

	local bestStat = Instance.new("StringValue")
	bestStat.Name = "Best"
	bestStat.Parent = stats

	stats.Parent = player
	refreshStats(player)
end

function Inventory.save(player: Player)
	local record = records[player]
	if not record or not store then
		return
	end
	local ok, err = pcall(function()
		(store :: DataStore):SetAsync(keyFor(player), record)
	end)
	if not ok and not storeWarned then
		storeWarned = true
		warn(`[CardSystem] profile save failed for {player.Name}: {err}`)
	end
end

function Inventory.release(player: Player)
	Inventory.save(player)
	records[player] = nil
end

function Inventory.get(player: Player): Record?
	return records[player]
end

function Inventory.getPity(player: Player): number
	local record = records[player]
	return if record then record.Pity else 0
end

function Inventory.setPity(player: Player, value: number)
	local record = records[player]
	if record then
		record.Pity = value
		record.Rolls += 1
	end
end

function Inventory.owns(player: Player, cardId: string): boolean
	local record = records[player]
	return record ~= nil and (record.Counts[cardId] or 0) > 0
end

function Inventory.award(player: Player, cardId: string): boolean
	local record = records[player]
	if not record or not Cards.get(cardId) then
		return false
	end

	record.Counts[cardId] = (record.Counts[cardId] or 0) + 1
	refreshStats(player)
	return true
end

function Inventory.equip(player: Player, slot: number, cardId: string?): (boolean, string?)
	local record = records[player]
	if not record then
		return false, "Profile still loading"
	end
	if slot ~= math.floor(slot) or slot < 1 or slot > Config.Hotbar.Slots then
		return false, "No such slot"
	end

	if cardId == nil then
		record.Equipped[tostring(slot)] = nil
		return true, nil
	end

	if not Cards.get(cardId) then
		return false, "Unknown card"
	end
	if (record.Counts[cardId] or 0) <= 0 then
		return false, "You do not own that"
	end

	for otherSlot, otherCard in record.Equipped do
		if otherCard == cardId then
			record.Equipped[otherSlot] = nil
		end
	end
	record.Equipped[tostring(slot)] = cardId
	return true, nil
end

function Inventory.firstFreeSlot(player: Player): number?
	local record = records[player]
	if not record then
		return nil
	end
	for slot = 1, Config.Hotbar.Slots do
		if record.Equipped[tostring(slot)] == nil then
			return slot
		end
	end
	return nil
end

function Inventory.bindShutdown()
	game:BindToClose(function()
		if RunService:IsStudio() then
			return
		end
		for _, player in Players:GetPlayers() do
			Inventory.save(player)
		end
		task.wait(1)
	end)
end

return Inventory

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local CardSystem = ReplicatedStorage:WaitForChild("CardSystem")
local Cards = require(CardSystem:WaitForChild("Cards"))
local Config = require(CardSystem:WaitForChild("Config"))
local RollLogic = require(CardSystem:WaitForChild("RollLogic"))
local Remotes = require(CardSystem:WaitForChild("Remotes")).get()

local Inventory = require(script:WaitForChild("Inventory"))

type PendingRoll = {
	Id: number,
	CardIds: { string },
}

local rng = Random.new()
local pending: { [Player]: PendingRoll } = {}
local nextRollAt: { [Player]: number } = {}
local nextRollId = 0

local function sync(player: Player)
	local snapshot = Inventory.snapshot(player)
	if snapshot then
		Remotes.ProfileSync:FireClient(player, snapshot)
	end
end

local function reject(player: Player, reason: string, retryAfter: number?)
	Remotes.RollRejected:FireClient(player, reason, retryAfter or 0)
end

local function onRequestRoll(player: Player)
	if not Inventory.get(player) then
		reject(player, "Still loading your profile, try again.")
		return
	end

	if pending[player] then
		reject(player, "Pick a card first.")
		return
	end

	local now = os.clock()
	local readyAt = nextRollAt[player] or 0
	if now < readyAt then
		reject(player, "Rolling too fast.", readyAt - now)
		return
	end

	nextRollAt[player] = now + Config.RollCooldown

	local hand, newPity = RollLogic.rollHand(rng, Inventory.getPity(player))
	Inventory.setPity(player, newPity)

	nextRollId += 1
	local cardIds = {}
	for _, card in hand do
		table.insert(cardIds, card.Id)
	end

	pending[player] = { Id = nextRollId, CardIds = cardIds }
	Remotes.RollResult:FireClient(player, nextRollId, cardIds)
end

local function onSelectCard(player: Player, rollId: any, slot: any)
	local roll = pending[player]
	local function refuse()
		Remotes.SelectionResult:FireClient(player, false, nil, nil)
	end

	if not roll then
		refuse()
		return
	end
	if typeof(rollId) ~= "number" or rollId ~= roll.Id then
		refuse()
		return
	end
	if typeof(slot) ~= "number" or slot ~= math.floor(slot) or slot < 1 or slot > #roll.CardIds then
		refuse()
		return
	end

	local cardId = roll.CardIds[slot]
	pending[player] = nil
	nextRollAt[player] = os.clock() + Config.RollCooldown

	if not Inventory.award(player, cardId) then
		warn(`[CardSystem] could not award "{cardId}" to {player.Name}`)
		refuse()
		return
	end

	Remotes.SelectionResult:FireClient(player, true, cardId, slot)
	sync(player)
end

local function onEquipCard(player: Player, slot: any, cardId: any)
	if typeof(slot) ~= "number" then
		return
	end
	if cardId ~= nil and typeof(cardId) ~= "string" then
		return
	end

	local ok, reason = Inventory.equip(player, slot, cardId)
	if not ok then
		Remotes.Notice:FireClient(player, reason or "Could not equip that", true)
		return
	end
	sync(player)
end

local function onPlayerAdded(player: Player)
	Inventory.load(player)
	sync(player)
end

local function onPlayerRemoving(player: Player)
	pending[player] = nil
	nextRollAt[player] = nil
	Inventory.release(player)
end

Remotes.RequestRoll.OnServerEvent:Connect(onRequestRoll)
Remotes.SelectCard.OnServerEvent:Connect(onSelectCard)
Remotes.EquipCard.OnServerEvent:Connect(onEquipCard)

Players.PlayerAdded:Connect(onPlayerAdded)
Players.PlayerRemoving:Connect(onPlayerRemoving)

for _, player in Players:GetPlayers() do
	task.spawn(onPlayerAdded, player)
end

Inventory.bindShutdown()

print(`[CardSystem] ready - {#Cards.List} cards, {Config.CardsPerRoll} per roll`)

local function clamp01(n: number): number
	return math.clamp(n, 0, 1)
end

local function easeOutCubic(t: number): number
	return 1 - (1 - t) ^ 3
end

local function easeOutBack(t: number): number
	local c1, c3 = 1.70158, 2.70158
	return 1 + c3 * (t - 1) ^ 3 + c1 * (t - 1) ^ 2
end

local function easeInOutCubic(t: number): number
	return if t < 0.5 then 4 * t ^ 3 else 1 - (-2 * t + 2) ^ 3 / 2
end

local function lerp(a: number, b: number, t: number): number
	return a + (b - a) * t
end

		Transparency = 1,
		Size = Config.CardSize,

	local front = buildFront(card, rarity)
	local back = buildBack()

	make("SurfaceGui", {
		Name = "FrontGui",
		Face = Enum.NormalId.Front,
		SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud,
		PixelsPerStud = Config.PixelsPerStud,

		Brightness = 1.9,
		MaxDistance = 250,

	make("SurfaceGui", {
		Name = "BackGui",
		Face = Enum.NormalId.Back,
		SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud,
		PixelsPerStud = Config.PixelsPerStud,
		LightInfluence = 0,
		Brightness = 1.6,
		MaxDistance = 250,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
		Parent = part,
	}, { back })

	local light = make("PointLight", {
		Color = rarity.GlowColor,
		Brightness = 0,
		Range = 10,
		Shadows = false,
		Parent = part,
	}) :: PointLight

	local glow = make("Highlight", {
		Name = "Hover",
		Adornee = part,
		FillTransparency = 1,
		OutlineColor = rarity.GlowColor,
		OutlineTransparency = 1,
		DepthMode = Enum.HighlightDepthMode.AlwaysOnTop,
		Parent = part,
	}) :: Highlight

	local attachment = make("Attachment", { Name = "SparkPoint", Parent = part }) :: Attachment
	local sparks = make("ParticleEmitter", {
		Name = "Sparks",
		Enabled = false,
		Color = ColorSequence.new(rarity.GlowColor),

		State = "entering",
		ExitStart = 0,
		ExitMode = "discarded",
		Revealed = false,
		Tilt = Vector2.zero,
		Hover = 0,
		OnRevealed = nil,
		Front = front,
		BackFace = back,
		Light = light,
		Glow = glow,
		Sparks = sparks,
	}, CardVisual)

	return (self :: any) :: CardVisual
end

function CardVisual.isInteractive(self: CardVisual): boolean
	return self.Revealed and self.State ~= "exiting"
end

function CardVisual.beginExit(self: CardVisual, mode: string)
	if self.State == "exiting" then
		return
	end
	self.State = "exiting"
	self.ExitMode = mode
	self.ExitStart = os.clock()
end

function CardVisual.isFinished(self: CardVisual): boolean
	return self.State == "exiting" and os.clock() - self.ExitStart > Config.ExitDuration
end

function CardVisual.update(self: CardVisual, dt: number, tiltTarget: Vector2, hovered: boolean)
	local now = os.clock()
	local elapsed = now - self.StartTime

	local enter = clamp01((elapsed - self.EnterDelay) / Config.EnterDuration)
	local enterEase = easeOutBack(enter)
	local flip = clamp01((elapsed - self.FlipDelay) / Config.FlipDuration)
	local flipEase = easeInOutCubic(flip)

	if flip >= 1 and not self.Revealed then
		self.Revealed = true
		if self.Rarity.Sparkle then
			self.Sparks:Emit(28 + self.Rarity.Rank * 8)
		end
		if self.OnRevealed then
			(self.OnRevealed :: any)(self.Slot)
		end
	end

	local interactive = self.Revealed and self.State ~= "exiting"
	local wantHover = if interactive and hovered then 1 else 0
	local smoothing = clamp01(dt * Config.TiltSmoothing)
	local tiltWeight = if interactive then 1 else 0
	self.Tilt = self.Tilt:Lerp(tiltTarget * tiltWeight, smoothing)
	self.Hover = lerp(self.Hover, wantHover, smoothing)

	local scale = lerp(Config.EnterScale, 1, math.min(enterEase, 1))
	scale *= lerp(1, Config.HoverScale, self.Hover)

	local rise = (1 - easeOutCubic(enter)) * -Config.EnterDrop + self.Hover * Config.HoverLift
	local alpha = 1 - enter
	local pushToCamera = self.Hover * Config.HoverPop

	if self.State == "exiting" then
		local exit = clamp01((now - self.ExitStart) / Config.ExitDuration)
		local exitEase = easeOutCubic(exit)
		alpha = math.max(alpha, exitEase)
		if self.ExitMode == "chosen" then
			pushToCamera += exitEase * 5
			scale *= lerp(1, 1.35, exitEase)
		else
			rise -= exitEase * 3
			scale *= lerp(1, 0.85, exitEase)
		end
	end

	local flipAngle = math.pi * (1 - flipEase)

	local cframe = CFrame.new(0, rise, 0)
		* self.BaseCFrame
		* CFrame.Angles(0, flipAngle, 0)
		* CFrame.Angles(self.Tilt.Y * Config.MaxTilt, self.Tilt.X * Config.MaxTilt, 0)
		* CFrame.new(0, 0, -pushToCamera)

	self.Part.CFrame = cframe
	self.Part.Size = Config.CardSize * scale
	self.Part.Transparency = alpha

	self.Front.GroupTransparency = alpha
	self.BackFace.GroupTransparency = alpha

	self.Light.Brightness = self.Rarity.GlowBrightness * flipEase * (1 - alpha)
	self.Glow.OutlineTransparency = 1 - self.Hover * 0.85
end

function CardVisual.destroy(self: CardVisual)
	self.Part:Destroy()
end

return CardVisual

function Hotbar.new(): Hotbar
	local playerGui = Players.LocalPlayer:WaitForChild("PlayerGui")

	local gui = make("ScreenGui", {
		Name = "CardHotbar",
		ResetOnSpawn = false,
		IgnoreGuiInset = true,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
		DisplayOrder = 8,
	}) :: ScreenGui

	local rowWidth = Config.Hotbar.Slots * L.HotbarSlot + (Config.Hotbar.Slots - 1) * L.HotbarGap

	local root = make("Frame", {

		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(0.5, 1),
		Position = UDim2.new(0.5, 0, 1, 0),
		Size = UDim2.fromOffset(rowWidth, L.HotbarBottom + L.HotbarSlot),
		Parent = gui,
	})

	local scale = make("UIScale", { Scale = 1, Parent = root }) :: UIScale

	local row = make("Frame", {
		Name = "Slots",
		BackgroundTransparency = 1,
		AnchorPoint = Vector2.new(0.5, 1),
		Position = UDim2.new(0.5, 0, 1, -L.HotbarBottom),
		Size = UDim2.fromOffset(rowWidth, L.HotbarSlot),
		Parent = root,
	}, {
		make("UIListLayout", {
			FillDirection = Enum.FillDirection.Horizontal,
			HorizontalAlignment = Enum.HorizontalAlignment.Center,
			VerticalAlignment = Enum.VerticalAlignment.Center,
			Padding = UDim.new(0, L.HotbarGap),
			SortOrder = Enum.SortOrder.LayoutOrder,
		}),
	})

	local slots: { SlotViews } = {}
	for index = 1, Config.Hotbar.Slots do
		slots[index] = buildSlot(index, row)
	end

	gui.Parent = playerGui

	local self = setmetatable({
		Gui = gui,
		Scale = scale,
		Slots = slots,
		Selected = 1,
		OnSlotActivated = nil,
	}, Hotbar)

	for index, slot in slots do
		slot.Button.MouseButton1Click:Connect(function()
			self:activate(index)
		end)
		slot.Button.MouseEnter:Connect(function()
			TweenService:Create(slot.Surface, TweenInfo.new(0.12), {
				Size = UDim2.fromScale(1.06, 1.06),
			}):Play()
		end)
		slot.Button.MouseLeave:Connect(function()
			TweenService:Create(slot.Surface, TweenInfo.new(0.12), {
				Size = UDim2.fromScale(1, 1),
			}):Play()
		end)
	end

	self:refresh()
	return (self :: any) :: Hotbar
end

function Hotbar.activate(self: Hotbar, index: number)
	self.Selected = index
	self:refresh()
	if self.OnSlotActivated then
		(self.OnSlotActivated :: any)(index, Profile.equippedAt(index))
	end
end

function Hotbar.setScale(self: Hotbar, scale: number)
	self.Scale.Scale = scale
end

function Hotbar.refresh(self: Hotbar)
	for index, views in self.Slots do
		local cardId = Profile.equippedAt(index)
		local card = if cardId then Cards.get(cardId) else nil

		if views.Icon then
			views.Icon:Destroy()
			views.Icon = nil
		end

		local accent = if card then Rarities.get(card.Rarity).Color else Color3.fromRGB(90, 96, 112)

		if card then
			views.Content.Visible = true
			views.NameLabel.Text = card.Name
			views.CountLabel.Text = `x{Profile.count(card.Id)}`

			views.Icon = Theme.iconFor(card.Name, card.Icon, accent, {
				AnchorPoint = Vector2.new(0.5, 0),
				Position = UDim2.fromScale(0.5, 0.14),
				Size = UDim2.fromScale(0.62, 0.44),
				Parent = views.Surface,
			})
		else
			views.Content.Visible = false
		end

		local selected = index == self.Selected

		local art = views.Art
		if art then

			art.ImageColor3 = if Theme.SlotArt.Tint and card then accent else Color3.new(1, 1, 1)

			art.ImageTransparency = if selected then 0 else 0.12
		else
			local stroke = views.Stroke
			if stroke then
				stroke.Color = accent
				stroke.Transparency = if card then 0 else 0.2
				stroke.Thickness = if selected then 5 else 3
			end
			views.Surface.BackgroundColor3 = if card
				then select(1, Theme.rarityBackdrop(accent))
				else Theme.Colors.SlotEmpty
		end

		views.Surface.Position = if selected
			then UDim2.new(0.5, 0, 0.5, -4)
			else UDim2.fromScale(0.5, 0.5)
	end
end

return Hotbar

function InventoryUI.showTooltip(self: InventoryUI, cardId: string)

function InventoryUI.moveTooltip(self: InventoryUI)
	if not self.Tooltip.Visible then
		return
	end
	local mouse = UserInputService:GetMouseLocation()
	local camera = workspace.CurrentCamera
	local viewport = camera.ViewportSize
	local size = self.Tooltip.AbsoluteSize

	local x = mouse.X + 16
	if x + size.X > viewport.X - 8 then
		x = mouse.X - size.X - 16
	end
	local y = mouse.Y + 12
	if y + size.Y > viewport.Y - 8 then
		y = viewport.Y - size.Y - 8
	end

	self.Tooltip.Position = UDim2.fromOffset(x, y)
end

function InventoryUI.hideTooltip(self: InventoryUI)
	self.Tooltip.Visible = false
	self.HoveredCard = nil
end

function InventoryUI.refresh(self: InventoryUI)
	for id, button in self.Tabs do
		local active = id == self.Category
		button.BackgroundTransparency = if active then 0 else 0.45
		local stroke = button:FindFirstChildOfClass("UIStroke")
		if stroke then
			stroke.Thickness = if active then 3.5 else 2
		end
	end

	local owned = Profile.ownedSorted(function(cardId: string): number
		local card = Cards.get(cardId)
		return if card then Rarities.get(card.Rarity).Rank else 0
	end)

	local shown = 0
	for _, entry in owned do
		local card = Cards.get(entry.Id)
		if not card then
			continue
		end
		if self.Category ~= Categories.AllId and card.Category ~= self.Category then
			continue
		end
		if self.Query ~= "" and not string.find(string.lower(card.Name), self.Query, 1, true) then
			continue
		end

		shown += 1
		local views = self.Slots[shown]
		if not views then
			break
		end

		local rarity = Rarities.get(card.Rarity)
		local top = select(1, Theme.rarityBackdrop(rarity.Color))

		views.CardId = card.Id
		views.Button.Visible = true
		views.Button.LayoutOrder = shown
		views.Surface.BackgroundColor3 = top
		views.Stroke.Color = rarity.Color
		views.Count.Text = `x{entry.Count}`

		if views.Icon then
			views.Icon:Destroy()
		end
		views.Icon = Theme.iconFor(card.Name, card.Icon, rarity.Color, {
			AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.fromScale(0.5, 0.44),
			Size = UDim2.fromScale(0.6, 0.5),
			Parent = views.Surface,
		})
	end

	for index = shown + 1, #self.Slots do
		local views = self.Slots[index]
		views.Button.Visible = false
		views.CardId = nil
	end

	self.EmptyLabel.Visible = shown == 0

	if self.HoveredCard and not Profile.owns(self.HoveredCard) then
		self:hideTooltip()
	elseif self.HoveredCard then
		self:showTooltip(self.HoveredCard)
	end
end

function InventoryUI.open(self: InventoryUI)
	self.Open = true
	self.Gui.Enabled = true
	self:refresh()

	local scale = make("UIScale", { Scale = 0.9, Parent = self.Root }) :: UIScale
	local tween = TweenService:Create(scale, TweenInfo.new(0.18, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
		Scale = 1,
	})
	tween.Completed:Connect(function()
		scale:Destroy()
	end)
	tween:Play()
end

function InventoryUI.close(self: InventoryUI)
	self.Open = false
	self.Gui.Enabled = false
	self:hideTooltip()
	if self.OnClosed then
		(self.OnClosed :: any)()
	end
end

function InventoryUI.toggle(self: InventoryUI)
	if self.Open then
		self:close()
	else
		self:open()
	end
end

return InventoryUI

local ReplicatedStorage = game:GetService("ReplicatedStorage")

export type Snapshot = {
	Counts: { [string]: number },
	Equipped: { string | boolean },
	Rolls: number,
}

local Profile = {}

local current: Snapshot = {
	Counts = {},
	Equipped = {},
	Rolls = 0,
}

local listeners: { (Snapshot) -> () } = {}

function Profile.get(): Snapshot
	return current
end

function Profile.set(snapshot: Snapshot)
	current = snapshot
	for _, listener in listeners do
		task.spawn(listener, snapshot)
	end
end

function Profile.onChanged(listener: (Snapshot) -> ()): () -> ()
	table.insert(listeners, listener)
	return function()
		local index = table.find(listeners, listener)
		if index then
			table.remove(listeners, index)
		end
	end
end

function Profile.count(cardId: string): number
	return current.Counts[cardId] or 0
end

function Profile.owns(cardId: string): boolean
	return Profile.count(cardId) > 0
end

function Profile.equippedAt(slot: number): string?
	local value = current.Equipped[slot]
	return if typeof(value) == "string" then value else nil
end

function Profile.slotOf(cardId: string): number?
	for slot = 1, Config.Hotbar.Slots do
		if Profile.equippedAt(slot) == cardId then
			return slot
		end
	end
	return nil
end

function Profile.ownedSorted(rank: (string) -> number): { { Id: string, Count: number } }
	local out = {}
	for cardId, count in current.Counts do
		if count > 0 then
			table.insert(out, { Id = cardId, Count = count })
		end
	end
	table.sort(out, function(a, b)
		local rankA, rankB = rank(a.Id), rank(b.Id)
		if rankA ~= rankB then
			return rankA > rankB
		end
		return a.Id < b.Id
	end)
	return out
end

return Profile

function RollUI.runCooldown(
	self: RollUI,
	seconds: number,
	isCurrent: () -> boolean,
	onFinished: (() -> ())?
)
	task.spawn(function()
		self:setEnabled(false)
		local remaining = seconds
		while remaining > 0 do
			if not isCurrent() then
				return
			end
			self:setCooldownText(string.format("%.1fs", remaining))
			remaining -= task.wait(0.1)
		end
		if not isCurrent() then
			return
		end
		self:setCooldownText(nil)
		self:setEnabled(true)
		if onFinished then
			onFinished()
		end
	end)
end

	self.BannerLabel.TextTransparency = 1

	local inInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
	TweenService:Create(banner, inInfo, { BackgroundTransparency = 0.05 }):Play()
	TweenService:Create(self.BannerLabel, inInfo, { TextTransparency = 0 }):Play()

	task.delay(2.4, function()
		local outInfo = TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
		TweenService:Create(self.BannerLabel, outInfo, { TextTransparency = 1 }):Play()
		local fade = TweenService:Create(banner, outInfo, { BackgroundTransparency = 1 })
		fade.Completed:Connect(function()
			banner.Visible = false
		end)
		fade:Play()
	end)
end

	self.Notice.Visible = true
	self.Notice.TextTransparency = 0

	task.delay(2.2, function()
		if self.Notice.Text ~= text then
			return
		end
		local fade = TweenService:Create(self.Notice, TweenInfo.new(0.3), { TextTransparency = 1 })
		fade.Completed:Connect(function()
			if self.Notice.Text == text then
				self.Notice.Visible = false
			end
		end)
		fade:Play()
	end)
end

function Showcase.show(self: Showcase, cardId: string)
	local card = Cards.get(cardId)
	if not card then
		return
	end
	local rarity = Rarities.get(card.Rarity)

	self:hide(true)

	local camera = workspace.CurrentCamera
	local base = camera.CFrame * CFrame.new(0, 0.2, -SHOW.Distance)
	self.Base = CFrame.lookAt(base.Position, camera.CFrame.Position)

	local part = make("Part", {
		Name = "ShowcaseItem",
		Anchored = true,
		CanCollide = false,
		CanTouch = false,
		CanQuery = false,
		CastShadow = false,
		Material = Enum.Material.SmoothPlastic,
		Color = Color3.fromRGB(14, 15, 20),
		Size = SHOW.Size,
		CFrame = self.Base,
		Parent = workspace,
	}) :: BasePart

	make("SurfaceGui", {
		Name = "FaceGui",
		Face = Enum.NormalId.Front,
		SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud,
		PixelsPerStud = Config.PixelsPerStud,
		LightInfluence = 0,
		Brightness = 1.6,
		MaxDistance = 250,
		Parent = part,
	}, { buildFace(card, rarity) })

	make("PointLight", {
		Color = rarity.GlowColor,
		Brightness = 3,
		Range = 16,
		Shadows = false,
		Parent = part,
	})

	local attachment = make("Attachment", { Name = "SparkPoint", Parent = part }) :: Attachment
	local sparks = make("ParticleEmitter", {
		Name = "Sparks",
		Color = ColorSequence.new(rarity.GlowColor),
		LightEmission = 1,
		Lifetime = NumberRange.new(0.6, 1.4),
		Speed = NumberRange.new(9, 20),
		SpreadAngle = Vector2.new(180, 180),
		Rate = 26,
		Rotation = NumberRange.new(0, 360),
		Size = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 0.7),
			NumberSequenceKeypoint.new(1, 0),
		}),
		Transparency = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 0.05),
			NumberSequenceKeypoint.new(1, 1),
		}),
		Parent = attachment,
	}) :: ParticleEmitter
	sparks:Emit(60)

	self.Part = part
	self.Sparks = sparks
	self.StartTime = os.clock()

	self.NameLabel.Text = card.Name
	self.RarityLabel.Text = rarity.Name
	self.RarityLabel.TextColor3 = rarity.TextColor
	self.OwnedLabel.Text = `Owned: {Profile.count(cardId)}x`
	self.TagLabel.Text = card.Description

	self.Gui.Enabled = true
	self.Open = true

	for _, label in { self.NameLabel, self.RarityLabel, self.OwnedLabel, self.TagLabel } do
		label.TextTransparency = 1
		TweenService:Create(label, TweenInfo.new(0.25), { TextTransparency = 0 }):Play()
	end

	self.Connection = RunService.RenderStepped:Connect(function()
		self:step()
	end)
end

function Showcase.step(self: Showcase)
	local part = self.Part
	if not part then
		return
	end
	local elapsed = os.clock() - self.StartTime
	local bob = math.sin(elapsed * SHOW.BobSpeed) * SHOW.BobHeight
	local spin = math.sin(elapsed * SHOW.SpinSpeed) * 0.25

	part.CFrame = CFrame.new(0, bob, 0) * self.Base * CFrame.Angles(0, spin, 0)
end

function Showcase.hide(self: Showcase, silent: boolean?)
	if self.Connection then
		self.Connection:Disconnect()
		self.Connection = nil
	end
	if self.Part then
		self.Part:Destroy()
		self.Part = nil
	end
	self.Sparks = nil

	local wasOpen = self.Open
	self.Open = false
	self.Gui.Enabled = false

	if wasOpen and not silent and self.OnClosed then
		(self.OnClosed :: any)()
	end
end

return Showcase

local function getControls(): any
	if cachedControls then
		return cachedControls
	end
	local scripts = player:FindFirstChild("PlayerScripts")
	local playerModule = scripts and scripts:FindFirstChild("PlayerModule")
	if not playerModule then
		return nil
	end

	local requireModule: (Instance) -> any = require :: any
	local ok, result = pcall(function()
		return requireModule(playerModule):GetControls()
	end)
	if ok then
		cachedControls = result
	end
	return cachedControls
end

local function applyMovementLock()
	local locked = next(movementLocks) ~= nil

	local controls = getControls()
	if controls then
		local ok = pcall(function()
			if locked then
				controls:Disable()
			else
				controls:Enable()
			end
		end)
		if ok then
			return
		end
		cachedControls = nil
	end

	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid then
		return
	end

	if locked then
		if savedWalkSpeed == nil then
			savedWalkSpeed = humanoid.WalkSpeed
			savedJumpPower = humanoid.JumpPower
		end
		humanoid.WalkSpeed = 0
		humanoid.JumpPower = 0
	elseif savedWalkSpeed then
		humanoid.WalkSpeed = savedWalkSpeed
		humanoid.JumpPower = savedJumpPower or humanoid.JumpPower
		savedWalkSpeed = nil
		savedJumpPower = nil
	end
end

local function setMovementLock(reason: string, locked: boolean)
	if not Config.LockMovementDuringCards then
		return
	end
	movementLocks[reason] = if locked then true else nil
	applyMovementLock()
end

local function playSound(id: string, pitch: number?)
	if id == "" then
		return
	end
	local sound = Instance.new("Sound")
	sound.SoundId = id
	sound.Volume = Config.SoundVolume
	sound.PlaybackSpeed = pitch or 1
	sound.PlayOnRemove = true
	sound.Parent = SoundService
	sound:Destroy()
end

local function setState(newState: State)
	state = newState
	stateToken += 1
	setMovementLock("cards", newState == "choosing" or newState == "resolving")
end

local function currentTokenGuard(): () -> boolean
	local token = stateToken
	return function()
		return token == stateToken
	end
end

local function refreshFilter()
	local parts: { Instance } = {}
	for _, visual in visuals do
		table.insert(parts, visual.Part)
	end
	raycastParams.FilterDescendantsInstances = parts
end

local function clearHand()
	for _, visual in visuals do
		visual:destroy()
	end
	table.clear(visuals)
	refreshFilter()
	if handFolder then
		handFolder:Destroy()
		handFolder = nil
	end
end

local function handOrigin(): CFrame
	return camera.CFrame * CFrame.new(0, Config.CameraHeightOffset, -Config.CameraDistance)
end

local function spawnHand(cardIds: { string })
	clearHand()
	camera = workspace.CurrentCamera

	local folder = Instance.new("Folder")
	folder.Name = HAND_FOLDER_NAME
	folder.Parent = workspace
	handFolder = folder

	local origin = handOrigin()
	local eye = camera.CFrame.Position
	local middle = (#cardIds + 1) / 2

	for slot, cardId in cardIds do
		local card = Cards.get(cardId)
		if not card then
			warn(`[CardSystem] server sent unknown card "{cardId}"`)
			continue
		end

		local offset = (slot - middle) * Config.CardSpacing
		local position = origin.Position + origin.RightVector * offset

		local visual = CardVisual.new(card, slot, CFrame.lookAt(position, eye), folder)

		visual.OnRevealed = function(revealedSlot: number)
			playSound(Config.Sounds.Reveal, 1 + (revealedSlot - 1) * Config.RevealPitchStep)
		end

		table.insert(visuals, visual)
	end

	refreshFilter()
	playSound(Config.Sounds.Roll)
end

local function cardUnderPoint(screenX: number, screenY: number): CardVisual.CardVisual?
	if #visuals == 0 then
		return nil
	end
	local ray = camera:ViewportPointToRay(screenX, screenY)
	local hit = workspace:Raycast(ray.Origin, ray.Direction * RAY_LENGTH, raycastParams)
	if not hit then
		return nil
	end
	for _, visual in visuals do
		if visual.Part == hit.Instance then
			return visual
		end
	end
	return nil
end

local function requestRoll()
	if state ~= "idle" then
		return
	end
	setState("requesting")
	ui:setEnabled(false)
	ui:setStatus("Dealing...")
	Remotes.RequestRoll:FireServer()

	local guard = currentTokenGuard()
	task.delay(5, function()
		if guard() then
			setState("idle")
			ui:setEnabled(true)
			ui:setStatus("No answer from the server, try again")
		end
	end)
end

local function selectCard(visual: CardVisual.CardVisual)
	if state ~= "choosing" or activeRollId == nil then
		return
	end
	if not visual:isInteractive() then
		return
	end

	setState("resolving")
	ui:setStatus("...")
	Remotes.SelectCard:FireServer(activeRollId, visual.Slot)
end

local function returnToIdle(delay: number)
	setState("cooldown")
	ui:runCooldown(delay, currentTokenGuard(), function()
		setState("idle")
		ui:setStatus(`Roll for {Config.CardsPerRoll} cards`)
	end)
end

RunService.RenderStepped:Connect(function(dt: number)
	if #visuals == 0 then
		return
	end

	camera = workspace.CurrentCamera
	local viewport = camera.ViewportSize
	local mouse = UserInputService:GetMouseLocation()

	local hovered: CardVisual.CardVisual? = nil
	if state == "choosing" then
		hovered = cardUnderPoint(mouse.X, mouse.Y)
	end

	for index = #visuals, 1, -1 do
		local visual = visuals[index]

		local screenPoint = camera:WorldToViewportPoint(visual.Part.Position)
		local dx = math.clamp((mouse.X - screenPoint.X) / (viewport.X * 0.5), -1, 1)
		local dy = math.clamp((mouse.Y - screenPoint.Y) / (viewport.Y * 0.5), -1, 1)

		local tilt = Vector2.new(dx, -dy)

		local isHovered = hovered == visual
		if not isHovered then
			tilt *= UNHOVERED_TILT
		end

		visual:update(dt, tilt, isHovered)

		if visual:isFinished() then
			visual:destroy()
			table.remove(visuals, index)
			refreshFilter()
		end
	end

	if #visuals == 0 and handFolder then
		handFolder:Destroy()
		handFolder = nil
	end
end)

local function repositionHand()
	if #visuals == 0 then
		return
	end
	local origin = handOrigin()
	local eye = camera.CFrame.Position
	local middle = (#visuals + 1) / 2
	for index, visual in visuals do
		local offset = (index - middle) * Config.CardSpacing
		local position = origin.Position + origin.RightVector * offset
		visual.BaseCFrame = CFrame.lookAt(position, eye)
	end
end
